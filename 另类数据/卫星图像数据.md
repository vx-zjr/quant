```python
def build_satellite_factor(parking_data, stock_returns, lookback=5):
    """
    构建卫星数据因子
    
    停车场车辆数 -> 零售企业营收
    """
    # 计算变化率
    parking_change = parking_data.pct_change(lookback)
    
    # 标准化
    factor = (parking_change - parking_change.mean()) / parking_change.std()
    
    # 与收益率合并
    factor_returns = pd.merge(
        stock_returns, factor, 
        left_index=True, right_index=True,
        how='inner'
    )
    
    return factor_returns

def correlate_satellite_returns(parking_df, revenue_df, company):
    """
    验证卫星数据与营收相关性
    """
    # 滞后处理（卫星数据通常提前1-2个月）
    parking_lagged = parking_df[company].shift(30)  # 30天滞后
    
    # 相关性
    corr = parking_lagged.corr(revenue_df[company])
    
    return {
        'correlation': corr,
        'p_value': stats.pearsonr(parking_lagged.dropna(), 
                                  revenue_df[company].loc[parking_lagged.dropna().index])[1]
    }
```

```python
def process_credit_card_data(transactions_df, category_mapping):
    """
    处理信用卡消费数据
    
    返回:
    - 消费增速
    - 市场占有率
    - 消费者偏好变化
    """
    # 按公司聚合
    company_spending = transactions_df.groupby('company')['amount'].sum()
    
    # 计算市场份额
    total_spending = company_spending.sum()
    market_share = company_spending / total_spending
    
    # 计算同比增速
    company_growth = company_spending.pct_change(365)
    
    # 分类统计
    category_spending = transactions_df.groupby('category')['amount'].agg(['sum', 'mean', 'count'])
    
    return {
        'market_share': market_share,
        'growth_rate': company_growth,
        'category_stats': category_spending
    }

def build_consumer_spending_factor(credit_data, stock_returns):
    """
    构建消费因子
    """
    # 公司消费增速
    growth = credit_data.groupby('date')['company'].apply(
        lambda x: x.pct_change()
    )
    
    # 因子标准化
    factor = (growth - growth.mean()) / growth.std()
    
    # 与股票收益合并
    merged = pd.merge(stock_returns, factor, on='date')
    
    # IC测试
    ic = stats.spearmanr(factor, stock_returns)[0]
    
    return {
        'factor': factor,
        'ic': ic,
        'data': merged
    }
```

```python
def build_sentiment_factor(tweets_df, lookback=5):
    """
    构建舆情因子
    """
    # 按日期聚合
    daily_sentiment = tweets_df.groupby('date')['sentiment'].agg(['mean', 'std', 'count'])
    
    # 累积因子
    sentiment_cumulative = daily_sentiment['mean'].rolling(lookback).sum()
    
    # 情绪分化因子
    sentiment_divergence = daily_sentiment['std'].rolling(lookback).mean()
    
    return {
        'sentiment': sentiment_cumulative,
        'divergence': sentiment_divergence,
        'volume': daily_sentiment['count']
    }
```

```python
def build_news_factor(news_df, stock_returns, lag=1):
    """
    构建新闻因子
    """
    # 情绪得分
    news_df['sentiment_score'] = news_df['sentiment'].map({
        'Bearish': -1, 
        'Neutral': 0, 
        'Bullish': 1
    }) * news_df['relevance']
    
    # 按日期聚合
    daily_news = news_df.groupby('date')['sentiment_score'].sum()
    
    # 滞后
    daily_news_lagged = daily_news.shift(lag)
    
    # 与收益合并
    merged = pd.merge(stock_returns, daily_news_lagged, on='date', how='inner')
    
    # IC计算
    ic = stats.spearmanr(merged['sentiment_score'], merged['return'])[0]
    
    return {
        'factor': daily_news_lagged,
        'ic': ic,
        'data': merged
    }
```

### 数据质量评估

```python
def evaluate_alt_data_quality(data_df):
    """
    评估另类数据质量
    """
    results = {}
    
    # 1. 覆盖度
    coverage = len(data_df.dropna()) / len(data_df)
    results['coverage'] = coverage
    
    # 2. 时效性
    time_diff = pd.Timestamp.now() - data_df['date'].max()
    results['timeliness'] = time_diff
    
    # 3. 频率一致性
    freq_analysis = data_df['date'].diff().value_counts()
    results['frequency'] = freq_analysis.mode()[0]
    
    # 4. 数据完整性
    missing_pct = data_df.isnull().sum() / len(data_df)
    results['completeness'] = 1 - missing_pct.mean()
    
    # 5. 异常值检测
    from scipy import stats
    z_scores = np.abs(stats.zscore(data_df.select_dtypes(include=[np.number])))
    results['outlier_pct'] = (z_scores > 3).sum().sum() / z_scores.size
    
    return results

def calculate_alt_data_value(data_df, target_returns):
    """
    计算另类数据价值
    """
    # 数据预处理
    clean_data = data_df.dropna()
    
    # IC分析
    ic_series = []
    for i in range(20, len(clean_data)):
        factor = clean_data['factor'].iloc[i-20:i]
        returns = target_returns.iloc[i-20:i]
        
        ic = stats.spearmanr(factor, returns)[0]
        ic_series.append(ic)
    
    results = {
        'ic_mean': np.mean(ic_series),
        'ic_std': np.std(ic_series),
        'ic_ir': np.mean(ic_series) / np.std(ic_series),
        'ic_positive_rate': np.mean([x > 0 for x in ic_series])
    }
    
    # 预测能力评估
    from sklearn.linear_model import LinearRegression
    
    X = clean_data['factor'].values.reshape(-1, 1)
    y = target_returns.loc[clean_data.index].values
    
    model = LinearRegression()
    model.fit(X[:-100], y[:-100])
    
    test_pred = model.predict(X[-100:])
    test_ic = stats.spearmanr(test_pred, y[-100:])[0]
    
    results['test_ic'] = test_ic
    
    return results
```

```python
class AltDataFetcher:
    """另类数据获取器"""
    
    def __init__(self, providers):
        self.providers = providers
    
    def fetch_satellite_data(self, location, date):
        """获取卫星数据"""
        provider = self.providers['satellite']
        return provider.get_image(location, date)
    
    def fetch_credit_card_data(self, company, start_date, end_date):
        """获取信用卡数据"""
        provider = self.providers['credit_card']
        return provider.get_spending_data(company, start_date, end_date)
    
    def fetch_news_sentiment(self, company, start_date, end_date):
        """获取新闻情绪"""
        provider = self.providers['news']
        return provider.get_sentiment(company, start_date, end_date)
    
    def aggregate_alt_signals(self, signals_dict):
        """
        聚合另类信号
        
        多个数据源的加权组合
        """
        # 计算每个信号权重
        weights = {
            'satellite': 0.3,
            'credit_card': 0.4,
            'news': 0.3
        }
        
        # 加权组合
        combined_signal = sum(
            signals_dict[key] * weight 
            for key, weight in weights.items()
        )
        
        return combined_signal
```

```python
def clean_alt_data(data_df):
    """
    另类数据清洗
    """
    # 缺失值处理
    data_df = data_df.fillna(method='ffill')
    data_df = data_df.fillna(0)
    
    # 异常值处理
    for col in data_df.select_dtypes(include=[np.number]).columns:
        q1 = data_df[col].quantile(0.25)
        q3 = data_df[col].quantile(0.75)
        iqr = q3 - q1
        
        lower = q1 - 1.5 * iqr
        upper = q3 + 1.5 * iqr
        
        data_df[col] = data_df[col].clip(lower, upper)
    
    # 标准化
    data_df = (data_df - data_df.mean()) / data_df.std()
    
    return data_df
```
