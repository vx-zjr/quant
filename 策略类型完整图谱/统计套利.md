## 一、统计套利概述

### 什么是统计套利
- **定义**: 利用资产间的统计关系进行套利
- **核心**: 均值回归、协整关系
- **特点**: 低风险、市场中性

```python
from statsmodels.tsa.stattools import coint, adfuller
import numpy as np

def test_cointegration(series1, series2):
    """
    协整检验
    
    H0: 两个序列不存在协整关系
    """
    # Engle-Granger两步法
    # 步骤1: OLS回归
    from sklearn.linear_model import LinearRegression
    reg = LinearRegression()
    reg.fit(series1.values.reshape(-1, 1), series2.values)
    residuals = series2 - reg.predict(series1.values.reshape(-1, 1))
    
    # 步骤2: 残差平稳性检验
    adf_result = adfuller(residuals)
    
    # Johansen检验
    from statsmodels.tsa.vector_ar.vecm import coint_johansen
    result = coint_johansen(
        np.column_stack([series1, series2]),
        det_order=0,
        k_ar_diff=1
    )
    
    return {
        'eg_statistic': adf_result[0],
        'eg_pvalue': adf_result[1],
        'trace_stat': result.lr1,
        'eigen_stat': result.lr2,
        'critical_values': result.cvt,
        'hedge_ratio': reg.coef_[0]
    }

def rolling_cointegration_test(series1, series2, window=60):
    """
    滚动协整检验
    """
    results = []
    
    for i in range(window, len(series1)):
        s1_window = series1.iloc[i-window:i]
        s2_window = series2.iloc[i-window:i]
        
        # 计算对冲比率
        hedge_ratio = np.cov(s1_window, s2_window)[0, 1] / np.var(s1_window)
        
        # 计算价差
        spread = s2_window - hedge_ratio * s1_window
        
        # ADF检验
        adf_result = adfuller(spread)
        
        results.append({
            'date': series1.index[i],
            'hedge_ratio': hedge_ratio,
            'adf_statistic': adf_result[0],
            'p_value': adf_result[1],
            'is_stationary': adf_result[1] < 0.05
        })
    
    return pd.DataFrame(results)
```

### 配对交易实现

```python
class PairsTradingStrategy:
    """配对交易策略"""
    
    def __init__(self, window=60, entry_threshold=2, exit_threshold=0.5):
        self.window = window
        self.entry_threshold = entry_threshold
        self.exit_threshold = exit_threshold
        self.hedge_ratio = None
        self.spread_mean = None
        self.spread_std = None
        
    def calculate_spread(self, series1, series2):
        """
        计算价差
        """
        # 滚动对冲比率
        hedge_ratios = []
        spreads = []
        
        for i in range(self.window, len(series1)):
            s1 = series1.iloc[i-self.window:i].values
            s2 = series2.iloc[i-self.window:i].values
            
            # OLS回归求对冲比率
            reg = LinearRegression()
            reg.fit(s1.reshape(-1, 1), s2)
            hedge_ratio = reg.coef_[0]
            hedge_ratios.append(hedge_ratio)
            
            # 当前价差
            spread = series2.iloc[i] - hedge_ratio * series1.iloc[i]
            spreads.append(spread)
        
        self.hedge_ratio = hedge_ratios[-1]
        
        # 滚动统计
        spread_series = pd.Series(spreads, index=series1.index[self.window:])
        self.spread_mean = spread_series.rolling(20).mean().iloc[-1]
        self.spread_std = spread_series.rolling(20).std().iloc[-1]
        
        return spread_series
    
    def calculate_zscore(self, spread):
        """
        计算Z-score
        """
        mean = spread.rolling(self.window).mean()
        std = spread.rolling(self.window).std()
        zscore = (spread - mean) / std
        
        return zscore
    
    def generate_signals(self, series1, series2):
        """
        生成交易信号
        """
        spread = self.calculate_spread(series1, series2)
        zscore = self.calculate_zscore(spread)
        
        signals = pd.DataFrame(index=spread.index)
        signals['spread'] = spread
        signals['zscore'] = zscore
        signals['signal'] = 0
        
        # 入场信号
        signals.loc[zscore > self.entry_threshold, 'signal'] = -1  # 做空价差
        signals.loc[zscore < -self.entry_threshold, 'signal'] = 1   # 做多价差
        
        # 出场信号
        signals.loc[abs(zscore) < self.exit_threshold, 'signal'] = 0
        
        return signals
    
    def backtest(self, series1, series2, initial_capital=100000):
        """
        配对交易回测
        """
        signals = self.generate_signals(series1, series2)
        
        capital = initial_capital
        position = 0  # 1: 做多价差, -1: 做空价差, 0: 空仓
        
        hedge_ratio = self.hedge_ratio
        entry_spread = 0
        
        trades = []
        equity_curve = [capital]
        
        for i in range(len(signals)):
            current_spread = signals['spread'].iloc[i]
            current_signal = signals['signal'].iloc[i]
            
            # 入场
            if position == 0 and current_signal != 0:
                position = current_signal
                entry_spread = current_spread
                entry_price1 = series1.iloc[i]
                entry_price2 = series2.iloc[i]
            
            # 出场
            elif position != 0:
                if (position == 1 and current_spread < entry_spread) or \
                   (position == -1 and current_spread > entry_spread):
                    # 平仓
                    pnl = position * (current_spread - entry_spread)
                    capital += pnl
                    trades.append({
                        'entry_spread': entry_spread,
                        'exit_spread': current_spread,
                        'pnl': pnl,
                        'position': position
                    })
                    position = 0
            
            # 更新权益
            if position != 0:
                current_pnl = position * (current_spread - entry_spread)
                portfolio_value = capital + current_pnl
            else:
                portfolio_value = capital
            
            equity_curve.append(portfolio_value)
        
        return {
            'final_capital': capital,
            'total_return': (capital - initial_capital) / initial_capital,
            'trades': pd.DataFrame(trades),
            'equity_curve': equity_curve,
            'signals': signals
        }
```

```python
def find_cointegrated_pairs(data_matrix):
    """
    找出所有协整配对
    
    参数:
    data_matrix: n_samples x n_assets
    """
    n_assets = data_matrix.shape[1]
    pairs = []
    
    for i in range(n_assets):
        for j in range(i + 1, n_assets):
            series1 = data_matrix[:, i]
            series2 = data_matrix[:, j]
            
            result = test_cointegration(series1, series2)
            
            # 如果协整
            if result['eg_pvalue'] < 0.05:
                pairs.append({
                    'asset1': i,
                    'asset2': j,
                    'hedge_ratio': result['hedge_ratio'],
                    'p_value': result['eg_pvalue']
                })
    
    # 按p值排序
    pairs.sort(key=lambda x: x['p_value'])
    
    return pairs

class MultiPairsStrategy:
    """多配对策略组合"""
    
    def __init__(self, pairs):
        self.pairs = pairs
        self.strategies = {}
        
        for pair in pairs:
            s1, s2 = pair['asset1'], pair['asset2']
            self.strategies[(s1, s2)] = PairsTradingStrategy()
    
    def generate_portfolio_signals(self, data_matrix):
        """
        生成组合信号
        """
        all_signals = []
        
        for pair in self.pairs:
            s1, s2 = pair['asset1'], pair['asset2']
            strategy = self.strategies[(s1, s2)]
            
            signals = strategy.generate_signals(
                pd.Series(data_matrix[:, s1]),
                pd.Series(data_matrix[:, s2])
            )
            
            all_signals.append(signals)
        
        return all_signals
    
    def portfolio_backtest(self, data_matrix):
        """
        组合回测
        """
        results = []
        
        for pair in self.pairs:
            s1, s2 = pair['asset1'], pair['asset2']
            strategy = self.strategies[(s1, s2)]
            
            result = strategy.backtest(
                pd.Series(data_matrix[:, s1]),
                pd.Series(data_matrix[:, s2])
            )
            
            results.append(result)
        
        # 汇总收益
        total_equity = sum([r['equity_curve'] for r in results])
        
        return {
            'individual_results': results,
            'total_return': (total_equity[-1] - 100000) / 100000,
            'total_equity': total_equity
        }
```

### 布林带回归策略
```python
def bollinger_band_reversion(prices, period=20, num_std=2):
    """
    布林带均值回归策略
    """
    signals = pd.DataFrame(index=prices.index)
    
    # 计算布林带
    signals['middle'] = prices.rolling(period).mean()
    signals['std'] = prices.rolling(period).std()
    signals['upper'] = signals['middle'] + num_std * signals['std']
    signals['lower'] = signals['middle'] - num_std * signals['std']
    
    # Z-score
    signals['zscore'] = (prices - signals['middle']) / signals['std']
    
    # 交易信号
    signals['position'] = 0
    
    # 价格触及下轨 -> 买入
    signals.loc[signals['zscore'] < -2, 'position'] = 1
    
    # 价格触及上轨 -> 卖出
    signals.loc[signals['zscore'] > 2, 'position'] = -1
    
    # 回归中轨 -> 平仓
    signals.loc[abs(signals['zscore']) < 0.5, 'position'] = 0
    
    return signals

# 改进版布林带策略
class AdvancedBollingerStrategy:
    """高级布林带策略"""
    
    def __init__(self, period=20, entry_std=2, exit_std=0.5):
        self.period = period
        self.entry_std = entry_std
        self.exit_std = exit_std
    
    def generate_signals(self, prices):
        """
        生成信号
        """
        # 基础布林带
        middle = prices.rolling(self.period).mean()
        std = prices.rolling(self.period).std()
        upper = middle + self.entry_std * std
        lower = middle - self.entry_std * std
        
        # Z-score
        zscore = (prices - middle) / std
        
        # 信号
        signals = pd.DataFrame(index=prices.index)
        signals['position'] = 0
        signals['entry_price'] = 0
        signals['zscore'] = zscore
        
        position = 0
        entry_price = 0
        
        for i in range(self.period, len(prices)):
            current_price = prices.iloc[i]
            current_zscore = zscore.iloc[i]
            
            if position == 0:
                # 入场
                if current_zscore < -self.entry_std:
                    position = 1
                    entry_price = current_price
                elif current_zscore > self.entry_std:
                    position = -1
                    entry_price = current_price
            
            elif position == 1:
                # 多头出场
                if current_zscore > -self.exit_std or current_price < entry_price * 0.97:
                    position = 0
            
            elif position == -1:
                # 空头出场
                if current_zscore < self.exit_std or current_price > entry_price * 1.03:
                    position = 0
            
            signals['position'].iloc[i] = position
        
        return signals
```

### RSI均值回归
```python
def rsi_mean_reversion(prices, period=14, oversold=30, overbought=70):
    """
    RSI均值回归策略
    """
    # 计算RSI
    delta = prices.diff()
    gain = (delta.where(delta > 0, 0)).rolling(period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(period).mean()
    
    rs = gain / loss
    rsi = 100 - (100 / (1 + rs))
    
    # 信号
    signals = pd.DataFrame(index=prices.index)
    signals['rsi'] = rsi
    signals['position'] = 0
    
    # RSI超卖 -> 买入
    signals.loc[rsi < oversold, 'position'] = 1
    
    # RSI超买 -> 卖出
    signals.loc[rsi > overbought, 'position'] = -1
    
    # RSI回归 -> 平仓
    signals.loc[(rsi > 50) & (signals['position'].shift(1) == 1), 'position'] = 0
    signals.loc[(rsi < 50) & (signals['position'].shift(1) == -1), 'position'] = 0
    
    return signals

# RSI-divergence策略
def rsi_divergence(prices, period=14):
    """
    RSI背离策略
    
    价格创新低但RSI没有创新低 -> 买入信号
    价格创新高但RSI没有创新高 -> 卖出信号
    """
    # 计算RSI
    delta = prices.diff()
    gain = (delta.where(delta > 0, 0)).rolling(period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(period).mean()
    rs = gain / loss
    rsi = 100 - (100 / (1 + rs))
    
    signals = pd.DataFrame(index=prices.index)
    
    # 检测背离
    # 使用简单的方法：比较RSI和价格的趋势
    price_trend = prices.diff(10)
    rsi_trend = rsi.diff(10)
    
    signals['bullish_divergence'] = (price_trend < 0) & (rsi_trend > 0)
    signals['bearish_divergence'] = (price_trend > 0) & (rsi_trend < 0)
    
    signals['position'] = 0
    signals.loc[signals['bullish_divergence'], 'position'] = 1
    signals.loc[signals['bearish_divergence'], 'position'] = -1
    
    return signals
```

```python
def factor_residual_strategy(prices, factor_data):
    """
    因子残差策略
    
    1. 对价格和因子做回归
    2. 交易残差（实际价格 - 预测价格）
    """
    from sklearn.linear_model import Ridge
    
    signals = pd.DataFrame(index=prices.index)
    
    # 滚动回归
    window = 60
    
    predictions = []
    residuals = []
    
    for i in range(window, len(prices)):
        X = factor_data.iloc[i-window:i].values
        y = prices.iloc[i-window:i].values
        
        model = Ridge(alpha=1.0)
        model.fit(X, y)
        
        # 预测
        pred = model.predict(factor_data.iloc[i:i+1].values)[0]
        residual = prices.iloc[i] - pred
        
        predictions.append(pred)
        residuals.append(residual)
    
    signals['prediction'] = [np.nan] * window + predictions
    signals['residual'] = [np.nan] * window + residuals
    
    # 残差Z-score
    residual_mean = pd.Series(residuals).rolling(20).mean()
    residual_std = pd.Series(residuals).rolling(20).std()
    signals['residual_zscore'] = (pd.Series(residuals) - residual_mean) / residual_std
    
    # 交易信号
    signals['position'] = 0
    signals.loc[signals['residual_zscore'] > 2, 'position'] = -1  # 价格高估
    signals.loc[signals['residual_zscore'] < -2, 'position'] = 1   # 价格低估
    
    return signals

# 多因子正交套利
def orthogonal_factor_arbitrage(factors, returns, lookback=60):
    """
    正交因子套利
    
    1. 因子正交化
    2. 残差均值回归
    """
    from sklearn.decomposition import PCA
    
    signals = pd.DataFrame(index=factors.index)
    
    for i in range(lookback, len(factors)):
        # 获取窗口数据
        window_factors = factors.iloc[i-lookback:i]
        window_returns = returns.iloc[i-lookback:i]
        
        # PCA正交化
        pca = PCA(n_components=min(5, window_factors.shape[1]))
        orthogonal_factors = pca.fit_transform(window_factors)
        
        # 回归
        reg = Ridge(alpha=1.0)
        reg.fit(orthogonal_factors, window_returns)
        
        # 预测收益
        current_factors = pca.transform(factors.iloc[i:i+1].values)
        predicted_return = reg.predict(current_factors)[0]
        
        # 实际收益
        actual_return = returns.iloc[i]
        
        # 残差
        residual = actual_return - predicted_return
        
        signals.loc[factors.index[i], 'predicted'] = predicted_return
        signals.loc[factors.index[i], 'actual'] = actual_return
        signals.loc[factors.index[i], 'residual'] = residual
    
    # 残差Z-score
    signals['residual_zscore'] = (
        signals['residual'] - signals['residual'].rolling(20).mean()
    ) / signals['residual'].rolling(20).std()
    
    # 信号
    signals['position'] = 0
    signals.loc[signals['residual_zscore'] < -2, 'position'] = 1
    signals.loc[signals['residual_zscore'] > 2, 'position'] = -1
    
    return signals
```

```python
class StatArbRiskControl:
    """统计套利风控"""
    
    def __init__(self, max_position_pct=0.1, max_loss_pct=0.02):
        self.max_position_pct = max_position_pct
        self.max_loss_pct = max_loss_pct
    
    def calculate_hedge_ratio(self, prices1, prices2, method='ols'):
        """
        计算对冲比率
        """
        if method == 'ols':
            reg = LinearRegression()
            reg.fit(prices1.values.reshape(-1, 1), prices2.values)
            return reg.coef_[0]
        
        elif method == 'rolling_ols':
            # 滚动OLS
            window = 60
            ratios = []
            
            for i in range(window, len(prices1)):
                reg = LinearRegression()
                reg.fit(
                    prices1.iloc[i-window:i].values.reshape(-1, 1),
                    prices2.iloc[i-window:i].values
                )
                ratios.append(reg.coef_[0])
            
            return ratios[-1]
        
        elif method == 'kalman':
            # Kalman滤波器
            from filterpy.kalman import KalmanFilter
            
            kf = KalmanFilter(dim_x=1, dim_z=1)
            kf.x = np.array([[1.0]])  # 初始对冲比率
            kf.F = np.array([[1.0]])
            kf.H = np.array([[1.0]])
            kf.R = np.array([[0.1]])  # 测量噪声
            kf.Q = np.array([[0.01]])  # 过程噪声
            
            ratios = []
            
            for i in range(len(prices1)):
                # 预测
                kf.predict()
                
                # 更新
                z = prices2.iloc[i] / prices1.iloc[i] if prices1.iloc[i] != 0 else 1
                kf.update(np.array([z]))
                
                ratios.append(kf.x[0, 0])
            
            return ratios[-1]
    
    def position_sizing(self, capital, spread_std, target_risk_pct=0.02):
        """
        仓位计算
        
        每笔交易风险敞口 = 目标风险比例
        """
        risk_amount = capital * target_risk_pct
        position_size = risk_amount / (2 * spread_std)  # 双向止损
        
        return {
            'risk_amount': risk_amount,
            'position_size': position_size,
            'max_loss': 2 * spread_std * position_size
        }
    
    def monitor_correlation(self, series1, series2, window=60, threshold=0.5):
        """
        监控相关性
        
        相关性下降 -> 可能均值回归失效
        """
        rolling_corr = series1.rolling(window).corr(series2)
        
        return {
            'current_correlation': rolling_corr.iloc[-1],
            'correlation_history': rolling_corr,
            'alert': rolling_corr.iloc[-1] < threshold
        }
```

```python
def correlation_break_alert(series1, series2, window=60, threshold=0.7):
    """
    相关性断裂预警
    
    当历史上相关的两个品种相关性突然下降时发出预警
    """
    rolling_corr = series1.rolling(window).corr(series2)
    
    # 计算相关性变化
    corr_change = rolling_corr.diff()
    corr_acceleration = corr_change.diff()
    
    alerts = []
    
    for i in range(window, len(rolling_corr)):
        current_corr = rolling_corr.iloc[i]
        
        # 历史平均相关性
        hist_avg = rolling_corr.iloc[i-window:i].mean()
        
        # 相关性断裂
        if current_corr < threshold and hist_avg > 0.8:
            alerts.append({
                'date': rolling_corr.index[i],
                'current_corr': current_corr,
                'hist_avg': hist_avg,
                'change': current_corr - hist_avg
            })
    
    return pd.DataFrame(alerts)
```

```python
class StatisticalArbitrageStrategy:

                        'p_value': result['eg_pvalue']

        return pairs[:self.max_pairs]
    
    def generate_signals(self, price_matrix, pairs):

            s2_idx = pair['asset2']
            
            series1 = pd.Series(price_matrix[:, s1_idx])
            series2 = pd.Series(price_matrix[:, s2_idx])
            
            # 计算价差
            spread = series2 - pair['hedge_ratio'] * series1
            
            # Z-score
            zscore = (spread - spread.rolling(self.window).mean()) / \
                     spread.rolling(self.window).std()
            
            signals = pd.DataFrame({
                'spread': spread,
                'zscore': zscore,
                'asset1': series1,
                'asset2': series2
            })
            
            signals_list.append(signals)
        
        return signals_list
    
    def execute_trades(self, signals_list, capital):
        """
        执行交易
        """
        portfolio_value = capital
        positions = {}  # {pair_id: position}
        entry_prices = {}
        
        for i in range(len(signals_list[0])):
            for pair_idx, signals in enumerate(signals_list):
                zscore = signals['zscore'].iloc[i]
                spread = signals['spread'].iloc[i]
                
                pair_key = f"pair_{pair_idx}"
                
                # 入场
                if pair_key not in positions:
                    if zscore > self.entry_threshold:
                        positions[pair_key] = -1  # 做空价差
                        entry_prices[pair_key] = spread
                    elif zscore < -self.entry_threshold:
                        positions[pair_key] = 1   # 做多价差
                        entry_prices[pair_key] = spread
                
                # 出场/止损
                elif pair_key in positions:
                    entry = entry_prices[pair_key]
                    
                    # 止损
                    if abs(spread - entry) > self.stop_loss * signals['spread'].rolling(20).std().iloc[i]:
                        # 平仓
                        pnl = positions[pair_key] * (spread - entry)
                        portfolio_value += pnl
                        del positions[pair_key]
                        del entry_prices[pair_key]
                    
                    # 回归出场
                    elif (positions[pair_key] == 1 and zscore > -self.exit_threshold) or \
                         (positions[pair_key] == -1 and zscore < self.exit_threshold):
                        pnl = positions[pair_key] * (spread - entry)
                        portfolio_value += pnl
                        del positions[pair_key]
                        del entry_prices[pair_key]
        
        return {
            'final_value': portfolio_value,
            'return': (portfolio_value - capital) / capital,
            'positions': positions
        }
    
    def backtest(self, price_matrix, asset_names, initial_capital=1000000):
        """
        回测
        """
        # 找配对
        pairs = self.find_pairs(price_matrix, asset_names)
        
        # 生成信号
        signals_list = self.generate_signals(price_matrix, pairs)
        
        # 执行交易
        result = self.execute_trades(signals_list, initial_capital)
        
        return {
            'pairs': pairs,
            'signals': signals_list,
            'results': result
        }
```

```python
def evaluate_stat_arb_strategy(trades_df, returns_series):
    """
    评估统计套利策略
    """
    results = {}
    
    # 基础指标
    results['total_trades'] = len(trades_df)
    results['winning_trades'] = (trades_df['pnl'] > 0).sum()
    results['win_rate'] = results['winning_trades'] / results['total_trades'] if results['total_trades'] > 0 else 0
    
    # 收益指标
    results['total_pnl'] = trades_df['pnl'].sum()
    results['avg_win'] = trades_df[trades_df['pnl'] > 0]['pnl'].mean()
    results['avg_loss'] = trades_df[trades_df['pnl'] < 0]['pnl'].mean()
    results['profit_factor'] = abs(trades_df['pnl'].sum() / trades_df[trades_df['pnl'] < 0]['pnl'].sum())
    
    # 风险指标
    results['max_drawdown'] = calculate_max_drawdown_from_trades(trades_df)
    results['sharpe_ratio'] = returns_series.mean() / returns_series.std() * np.sqrt(252)
    
    # 持续性指标
    results['avg_holding_period'] = trades_df['period'].mean()
    results['correlation_stability'] = calculate_correlation_stability(returns_series)
    
    return results

def calculate_max_drawdown_from_trades(trades_df):
    """从交易记录计算最大回撤"""
    cumulative = trades_df['pnl'].cumsum()
    peak = cumulative.cummax()
    drawdown = cumulative - peak
    
    return drawdown.min()

def calculate_correlation_stability(returns_series, window=20):
    """计算相关性稳定性"""
    rolling_corr = returns_series.rolling(window).apply(lambda x: x.autocorr())
    
    return {
        'mean_autocorr': rolling_corr.mean(),
        'std_autocorr': rolling_corr.std(),
        'stability_score': 1 - rolling_corr.std() / abs(rolling_corr.mean())
    }
```
