## 二、期权定价与Greeks

### Black-Scholes定价
```python
import numpy as np
from scipy.stats import norm

def black_scholes_price(S, K, T, r, sigma, option_type='call'):
    """
    Black-Scholes期权定价
    
    参数:
    S: 标的资产当前价格
    K: 行权价
    T: 到期时间(年)
    r: 无风险利率
    sigma: 波动率
    option_type: 'call' 或 'put'
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    if option_type == 'call':
        price = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
    else:
        price = K * np.exp(-r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)
    
    return price

def black_scholes_greeks(S, K, T, r, sigma, option_type='call'):
    """
    计算期权Greeks
    """
    d1 = (np.log(S / K) + (r + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    
    # Delta
    if option_type == 'call':
        delta = norm.cdf(d1)
    else:
        delta = norm.cdf(d1) - 1
    
    # Gamma (Call和Put相同)
    gamma = norm.pdf(d1) / (S * sigma * np.sqrt(T))
    
    # Vega (Call和Put相同)
    vega = S * norm.pdf(d1) * np.sqrt(T) / 100  # 每1%波动率
    
    # Theta
    if option_type == 'call':
        theta = (-S * norm.pdf(d1) * sigma / (2 * np.sqrt(T)) 
                 - r * K * np.exp(-r * T) * norm.cdf(d2)) / 365
    else:
        theta = (-S * norm.pdf(d1) * sigma / (2 * np.sqrt(T)) 
                 + r * K * np.exp(-r * T) * norm.cdf(-d2)) / 365
    
    # Rho
    if option_type == 'call':
        rho = K * T * np.exp(-r * T) * norm.cdf(d2) / 100
    else:
        rho = -K * T * np.exp(-r * T) * norm.cdf(-d2) / 100
    
    return {
        'delta': delta,
        'gamma': gamma,
        'vega': vega,
        'theta': theta,
        'rho': rho,
        'd1': d1,
        'd2': d2
    }
```

### Monte Carlo定价
```python
def monte_carlo_option_price(S, K, T, r, sigma, n_paths=100000, n_steps=100):
    """
    Monte Carlo期权定价
    """
    dt = T / n_steps
    discount = np.exp(-r * T)
    
    # 模拟路径
    Z = np.random.standard_normal((n_paths, n_steps))
    drift = (r - 0.5 * sigma**2) * dt
    diffusion = sigma * np.sqrt(dt) * Z
    
    log_returns = drift + diffusion
    log_prices = np.cumsum(log_returns, axis=1)
    prices = S * np.exp(log_prices)
    
    # 欧式期权
    payoff_call = np.maximum(prices[:, -1] - K, 0)
    payoff_put = np.maximum(K - prices[:, -1], 0)
    
    price_call = discount * np.mean(payoff_call)
    price_put = discount * np.mean(payoff_put)
    
    # 标准误差
    se_call = np.std(payoff_call) / np.sqrt(n_paths)
    se_put = np.std(payoff_put) / np.sqrt(n_paths)
    
    return {
        'call_price': price_call,
        'put_price': price_put,
        'call_se': se_call,
        'put_se': se_put
    }

def asian_option_price(S, K, T, r, sigma, n_paths=100000, n_steps=100):
    """
    亚式期权定价
    """
    dt = T / n_steps
    
    # 模拟路径
    Z = np.random.standard_normal((n_paths, n_steps))
    drift = (r - 0.5 * sigma**2) * dt
    diffusion = sigma * np.sqrt(dt) * Z
    
    log_returns = drift + diffusion
    log_prices = np.cumsum(log_returns, axis=1)
    prices = S * np.exp(log_prices)
    
    # 平均价格亚式期权
    avg_prices = np.mean(prices, axis=1)
    
    payoff_call = np.maximum(avg_prices - K, 0)
    payoff_put = np.maximum(K - avg_prices, 0)
    
    discount = np.exp(-r * T)
    
    return {
        'call_price': discount * np.mean(payoff_call),
        'put_price': discount * np.mean(payoff_put)
    }
```

### 波动率套利
```python
class VolatilityArbitrage:
    """波动率套利策略"""
    
    def __init__(self, lookback=30):
        self.lookback = lookback
        self.historical_vols = {}
    
    def calculate_historical_vol(self, prices, window=30):
        """
        计算历史波动率
        """
        returns = prices.pct_change().dropna()
        historical_vol = returns.rolling(window).std() * np.sqrt(252)
        
        return historical_vol
    
    def detect_volatility_signal(self, symbol, option_data, prices):
        """
        检测波动率交易信号
        
        信号类型:
        - IV > HV: 波动率高估，卖出期权
        - IV < HV: 波动率低估，买入期权
        """
        # 计算隐含波动率
        implied_vol = option_data['implied_vol'].iloc[-1]
        
        # 计算历史波动率
        historical_vol = self.calculate_historical_vol(prices)
        hv = historical_vol.iloc[-1]
        
        # 波动率价差
        vol_spread = implied_vol - hv
        
        # Z-score
        historical_spreads = implied_vol - historical_vol.dropna()
        z_score = (vol_spread - historical_spreads.mean()) / historical_spreads.std()
        
        # 信号
        signal = 0
        if z_score > 2 and implied_vol > hv:
            signal = -1  # IV高估，卖出
        elif z_score < -2 and implied_vol < hv:
            signal = 1   # IV低估，买入
        
        return {
            'implied_vol': implied_vol,
            'historical_vol': hv,
            'vol_spread': vol_spread,
            'z_score': z_score,
            'signal': signal
        }
    
    def calculate_position_greeks(self, options_positions):
        """
        计算组合Greeks
        """
        portfolio_greeks = {
            'delta': 0,
            'gamma': 0,
            'vega': 0,
            'theta': 0
        }
        
        for position in options_positions:
            greeks = black_scholes_greeks(
                position['S'], position['K'], position['T'],
                position['r'], position['sigma'],
                position['option_type']
            )
            
            # 考虑仓位方向和数量
            multiplier = position['quantity'] * position['direction']
            
            portfolio_greeks['delta'] += greeks['delta'] * multiplier
            portfolio_greeks['gamma'] += greeks['gamma'] * multiplier
            portfolio_greeks['vega'] += greeks['vega'] * multiplier
            portfolio_greeks['theta'] += greeks['theta'] * multiplier
        
        return portfolio_greeks
    
    def delta_hedge(self, portfolio_greeks, target_delta=0):
        """
        Delta对冲
        """
        current_delta = portfolio_greeks['delta']
        delta_to_hedge = target_delta - current_delta
        
        # 需要买入/卖出的标的数量
        # 假设标的每变动1元，Delta变动1
        shares_to_trade = -delta_to_hedge
        
        return shares_to_trade
```

```python
def straddle_strategy(S, K, T, r, sigma, option_price):
    """
    Straddle策略分析
    
    同时买入相同行权价的Call和Put
    """
    # 理论价格
    call_price = black_scholes_price(S, K, T, r, sigma, 'call')
    put_price = black_scholes_price(S, K, T, r, sigma, 'put')
    
    # 实际价格
    actual_call = option_price['call']
    actual_put = option_price['put']
    
    # 波动率套利信号
    implied_vol_call = calculate_implied_vol(S, K, T, r, actual_call, 'call')
    implied_vol_put = calculate_implied_vol(S, K, T, r, actual_put, 'put')
    
    avg_iv = (implied_vol_call + implied_vol_put) / 2
    
    # 策略分析
    breakeven_lower = K - (call_price + put_price)
    breakeven_upper = K + (call_price + put_price)
    
    # Greeks分析
    greeks = {
        'delta': black_scholes_greeks(S, K, T, r, sigma, 'call')['delta'] + 
                 black_scholes_greeks(S, K, T, r, sigma, 'put')['delta'],
        'gamma': black_scholes_greeks(S, K, T, r, sigma, 'call')['gamma'] + 
                 black_scholes_greeks(S, K, T, r, sigma, 'put')['gamma'],
        'vega': black_scholes_greeks(S, K, T, r, sigma, 'call')['vega'] + 
                black_scholes_greeks(S, K, T, r, sigma, 'put')['vega'],
        'theta': black_scholes_greeks(S, K, T, r, sigma, 'call')['theta'] + 
                 black_scholes_greeks(S, K, T, r, sigma, 'put')['theta']
    }
    
    return {
        'theoretical_price': call_price + put_price,
        'actual_price': actual_call + actual_put,
        'breakeven_points': [breakeven_lower, breakeven_upper],
        'avg_implied_vol': avg_iv,
        'greeks': greeks,
        'signal': 'buy' if avg_iv > sigma else 'sell' if avg_iv < sigma else 'hold'
    }
```

### Iron Condor策略
```python
class IronCondorStrategy:
    """铁蝶式价差(Iron Condor)策略"""
    
    def __init__(self, S, T, r, sigma):
        self.S = S  # 当前标的价格
        self.T = T  # 到期时间
        self.r = r  # 无风险利率
        self.sigma = sigma  # 波动率
    
    def construct_iron_condor(self, put_lower_k, put_upper_k, call_lower_k, call_upper_k):
        """
        构建Iron Condor
        
        买入Put (低行权价)
        卖出Put (中间行权价)
        卖出Call (中间行权价)
        买入Call (高行权价)
        """
        # 买入低行权价Put
        long_put_price = black_scholes_price(self.S, put_lower_k, self.T, self.r, self.sigma, 'put')
        
        # 卖出中间行权价Put
        short_put_price = black_scholes_price(self.S, put_upper_k, self.T, self.r, self.sigma, 'put')
        
        # 卖出中间行权价Call
        short_call_price = black_scholes_price(self.S, call_lower_k, self.T, self.r, self.sigma, 'call')
        
        # 买入高行权价Call
        long_call_price = black_scholes_price(self.S, call_upper_k, self.T, self.r, self.sigma, 'call')
        
        # 净收入
        net_credit = short_put_price + short_call_price - long_put_price - long_call_price
        
        return {
            'long_put': {'strike': put_lower_k, 'price': long_put_price, 'type': 'long'},
            'short_put': {'strike': put_upper_k, 'price': short_put_price, 'type': 'short'},
            'short_call': {'strike': call_lower_k, 'price': short_call_price, 'type': 'short'},
            'long_call': {'strike': call_upper_k, 'price': long_call_price, 'type': 'long'},
            'net_credit': net_credit
        }
    
    def analyze_profit_loss(self, condor):
        """
        分析盈亏
        """
        # 最大盈利 = 净权利金
        max_profit = condor['net_credit']
        
        # 最大亏损 = (宽跨价差 - 净权利金)
        put_width = condor['short_put']['strike'] - condor['long_put']['strike']
        call_width = condor['long_call']['strike'] - condor['short_call']['strike']
        max_loss = max(put_width, call_width) - condor['net_credit']
        
        # 盈利平衡点
        upper_breakeven = condor['short_call']['strike'] + condor['net_credit']
        lower_breakeven = condor['short_put']['strike'] - condor['net_credit']
        
        return {
            'max_profit': max_profit,
            'max_loss': max_loss,
            'upper_breakeven': upper_breakeven,
            'lower_breakeven': lower_breakeven,
            'risk_reward_ratio': max_profit / max_loss if max_loss > 0 else 0
        }
    
    def probability_analysis(self, condor, n_simulations=100000):
        """
        概率分析
        """
        # 假设到期价格服从对数正态分布
        import numpy as np
        
        dt = self.T
        mu = self.r - 0.5 * self.sigma**2
        
        # 模拟到期价格
        Z = np.random.standard_normal(n_simulations)
        final_prices = self.S * np.exp((mu + 0.5 * self.sigma**2) * dt + 
                                        self.sigma * np.sqrt(dt) * Z)
        
        # 计算盈利概率
        lower = condor['lower_breakeven']
        upper = condor['upper_breakeven']
        
        profit_prob = np.mean((final_prices >= lower) & (final_prices <= upper))
        full_profit_prob = np.mean((final_prices >= condor['short_put']['strike']) & 
                                    (final_prices <= condor['short_call']['strike']))
        loss_prob = 1 - profit_prob
        
        return {
            'profit_probability': profit_prob,
            'full_profit_probability': full_profit_prob,
            'loss_probability': loss_prob,
            'expected_value': profit_prob * condor['net_credit'] - loss_prob * (max_loss)
        }
```

```python
class GreeksRiskManager:
    """Greeks风险管理"""
    
    def __init__(self, target_delta=0, target_gamma=0, target_vega=0):
        self.targets = {
            'delta': target_delta,
            'gamma': target_gamma,
            'vega': target_vega
        }
    
    def calculate_portfolio_greeks(self, positions, market_data):
        """
        计算组合Greeks
        """
        portfolio = {
            'delta': 0,
            'gamma': 0,
            'vega': 0,
            'theta': 0
        }
        
        for pos in positions:
            S = market_data.loc[pos['symbol'], 'price']
            K = pos['strike']
            T = pos['time_to_expiry']
            r = pos['risk_free_rate']
            sigma = market_data.loc[pos['symbol'], 'volatility']
            
            greeks = black_scholes_greeks(S, K, T, r, sigma, pos['option_type'])
            
            multiplier = pos['quantity'] * (1 if pos['direction'] == 'long' else -1)
            
            portfolio['delta'] += greeks['delta'] * multiplier * pos['contract_size']
            portfolio['gamma'] += greeks['gamma'] * multiplier * pos['contract_size']
            portfolio['vega'] += greeks['vega'] * multiplier * pos['contract_size']
            portfolio['theta'] += greeks['theta'] * multiplier * pos['contract_size']
        
        return portfolio
    
    def delta_hedge_suggestion(self, portfolio_greeks, underlying_price):
        """
        Delta对冲建议
        """
        current_delta = portfolio_greeks['delta']
        target_delta = self.targets['delta']
        
        delta_to_hedge = target_delta - current_delta
        
        # 标的数量
        shares_needed = -delta_to_hedge  # 卖出Delta等价于买入标的
        
        return {
            'action': 'buy' if shares_needed > 0 else 'sell',
            'shares': abs(shares_needed),
            'cost': abs(shares_needed) * underlying_price
        }
    
    def vega_hedge_suggestion(self, portfolio_greeks, option_chain):
        """
        Vega对冲建议
        
        通过买入/卖出期权来调整组合Vega
        """
        current_vega = portfolio_greeks['vega']
        target_vega = self.targets['vega']
        
        vega_to_hedge = target_vega - current_vega
        
        # 找到合适期限的期权
        atm_options = option_chain[option_chain['moneyness'].between(0.95, 1.05)]
        
        suggestions = []
        for _, opt in atm_options.iterrows():
            opt_vega = black_scholes_greeks(
                opt['S'], opt['K'], opt['T'], opt['r'], opt['sigma'], opt['type']
            )['vega']
            
            if abs(opt_vega) > 0:
                contracts_needed = -vega_to_hedge / opt_vega
                suggestions.append({
                    'strike': opt['K'],
                    'expiry': opt['expiry'],
                    'type': opt['type'],
                    'contracts': abs(contracts_needed),
                    'direction': 'buy' if vega_to_hedge < 0 else 'sell'
                })
        
        return suggestions
    
    def generate_rebalancing_orders(self, positions, market_data):
        """
        生成再平衡订单
        """
        portfolio_greeks = self.calculate_portfolio_greeks(positions, market_data)
        
        orders = []
        
        # Delta对冲
        delta_order = self.delta_hedge_suggestion(
            portfolio_greeks, 
            market_data.iloc[0]['price']
        )
        if abs(delta_order['shares']) > 0:
            orders.append({
                'type': 'delta_hedge',
                **delta_order
            })
        
        # Vega对冲
        vega_orders = self.vega_hedge_suggestion(
            portfolio_greeks,
            self.get_option_chain()
        )
        orders.extend(vega_orders)
        
        return orders
```

```python
import numpy as np

def garch_volatility_forecast(returns, p=1, q=1):
    """
    GARCH(p,q)波动率预测
    """
    # 拟合GARCH模型
    model = arch_model(returns, vol='Garch', p=p, q=q)

        predicted_prices = []
        
        for i in range(len(K)):
            price = black_scholes_price(S, K[i], T[i], r, sigma, 'call')
            predicted_prices.append(price)
        
        mse = np.mean((np.array(market_prices) - np.array(predicted_prices))**2)
        return mse
    
    # 优化波动率参数
    result = minimize(objective, [0.2], args=(S, K, T, r, market_prices))
    
    return {
        'implied_vol': result.x[0],
        'model_price': black_scholes_price(S, np.mean(K), np.mean(T), r, result.x[0], 'call')
    }

def realized_volatility(returns, window=30):
    """
    已实现波动率
    """
    # 5分钟收益率
    rv = np.sqrt((returns**2).sum() * 252 * 78)  # 每天78个5分钟
    
    # Parkinson
    parkinson = np.sqrt((1/(4*np.log(2))) * (np.log(returns.high/returns.low)**2).sum() * 252)
    
    # Garman-Klass
    gk = np.sqrt(0.5 * (np.log(returns.high/returns.low)**2).sum() 
                 - (2*np.log(2)-1) * (np.log(returns.close/returns.open)**2).sum() * 252)
    
    # Rogers-Satchell
    rs = np.sqrt((np.log(returns.high/returns.close) * np.log(returns.high/returns.open) 
                  + np.log(returns.low/returns.close) * np.log(returns.low/returns.open)).sum() * 252)
    
    return {
        'realized_vol': rv,
        'parkinson': parkinson,
        'garman_klass': gk,
        'rogers_satchell': rs
    }
```

```python
class OptionsQuantStrategy:
    """期权量化策略"""
    
    def __init__(self, config):
        self.capital = config.get('capital', 1000000)
        self.max_position_pct = config.get('max_position_pct', 0.1)
        self.target_delta = config.get('target_delta', 0)
        self.max_loss_pct = config.get('max_loss_pct', 0.02)
    
    def signal_generation(self, market_data, option_chain):
        """
        生成交易信号
        """
        signals = []
        
        # 波动率交易信号
        for _, opt in option_chain.iterrows():
            vol_signal = self.volatility_signal(opt, market_data)
            
            if vol_signal['signal'] != 'hold':
                signals.append({
                    'type': 'volatility',
                    'option': opt,
                    'signal': vol_signal
                })
        
        # 价差交易信号
        spread_signals = self.spread_signals(option_chain)
        signals.extend(spread_signals)
        
        return signals
    
    def volatility_signal(self, option, market_data):
        """
        波动率信号
        """
        # 计算隐含波动率
        iv = calculate_implied_vol(
            option['S'], option['K'], option['T'],
            option['r'], option['price'], option['type']
        )
        
        # 计算历史波动率
        hv = realized_volatility(market_data['returns'])['realized_vol']
        
        # 计算HV rolling
        hv_rolling = market_data['returns'].rolling(30).std() * np.sqrt(252)
        hv_current = hv_rolling.iloc[-1]
        
        # 信号
        z_score = (iv - hv_current) / hv_rolling.std()
        
        if z_score > 2:
            return {'signal': 'sell_vol', 'iv': iv, 'hv': hv_current, 'z_score': z_score}
        elif z_score < -2:
            return {'signal': 'buy_vol', 'iv': iv, 'hv': hv_current, 'z_score': z_score}
        else:
            return {'signal': 'hold', 'iv': iv, 'hv': hv_current, 'z_score': z_score}
    
    def spread_signals(self, option_chain):
        """
        价差交易信号
        """
        signals = []
        
        # 寻找Iron Condor机会
        atm_options = option_chain[
            (option_chain['moneyness'] > 0.95) & 
            (option_chain['moneyness'] < 1.05)
        ]
        
        for expiry in atm_options['expiry'].unique():
            expiry_options = atm_options[atm_options['expiry'] == expiry]
            
            # 选择OTM Put和OTM Call
            otm_puts = expiry_options[expiry_options['moneyness'] < 1].sort_values('strike')
            otm_calls = expiry_options[expiry_options['moneyness'] > 1].sort_values('strike')
            
            if len(otm_puts) >= 2 and len(otm_calls) >= 2:
                # 构建Iron Condor
                condor = self.construct_iron_condor(
                    otm_puts.iloc[0]['strike'],
                    otm_puts.iloc[1]['strike'],
                    otm_calls.iloc[0]['strike'],
                    otm_calls.iloc[1]['strike']
                )
                
                analysis = self.analyze_profit_loss(condor)
                
                # 如果盈利概率高则信号
                if analysis['profit_probability'] > 0.7:
                    signals.append({
                        'type': 'iron_condor',
                        'condor': condor,
                        'analysis': analysis
                    })
        
        return signals
    
    def execute_trades(self, signals, current_positions):
        """
        执行交易
        """
        orders = []
        
        for signal in signals:
            if signal['type'] == 'volatility':
                # 波动率交易
                opt = signal['option']
                vol_signal = signal['signal']
                
                # 仓位大小
                position_size = self.capital * self.max_position_pct / opt['price']
                
                if vol_signal['signal'] == 'buy_vol':
                    orders.append({
                        'action': 'buy',
                        'option': opt,
                        'quantity': int(position_size)
                    })
                elif vol_signal['signal'] == 'sell_vol':
                    orders.append({
                        'action': 'sell',
                        'option': opt,
                        'quantity': int(position_size)
                    })
            
            elif signal['type'] == 'iron_condor':
                condor = signal['condor']
                
                # 卖出铁蝶式价差
                orders.append({'action': 'sell', 'strike': condor['short_put']['strike'], 'type': 'put', 'quantity': 1})
                orders.append({'action': 'sell', 'strike': condor['short_call']['strike'], 'type': 'call', 'quantity': 1})
                orders.append({'action': 'buy', 'strike': condor['long_put']['strike'], 'type': 'put', 'quantity': 1})
                orders.append({'action': 'buy', 'strike': condor['long_call']['strike'], 'type': 'call', 'quantity': 1})
        
        return orders
    
    def risk_management(self, positions, market_data):
        """
        风险管理
        """
        # 计算组合Greeks
        portfolio_greeks = self.calculate_portfolio_greeks(positions, market_data)
        
        # 检查止损
        total_pnl = sum([p['pnl'] for p in positions])
        if abs(total_pnl) > self.capital * self.max_loss_pct:
            # 需要平仓
            return {'action': 'stop_loss', 'reason': 'max_loss_exceeded'}
        
        # 生成对冲订单
        rebalance_orders = self.generate_rebalancing_orders(positions, market_data)
        
        return {
            'greeks': portfolio_greeks,
            'rebalance_orders': rebalance_orders
        }
```
