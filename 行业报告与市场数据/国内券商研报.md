## 六、研报自动化获取

### Python获取研报
```python
import requests
import pandas as pd

class ReportDownloader:
    """研报下载器"""
    
    def __init__(self, api_key):
        self.api_key = api_key
        self.base_url = "https://api.report.com"
    
    def search_reports(self, keyword, start_date, end_date):
        """搜索研报"""
        url = f"{self.base_url}/search"
        params = {
            "keyword": keyword,
            "start_date": start_date,
            "end_date": end_date,
            "page_size": 50
        }
        headers = {"Authorization": f"Bearer {self.api_key}"}
        
        response = requests.get(url, params=params, headers=headers)
        return response.json()
    
    def download_report(self, report_id, save_path):
        """下载研报"""
        url = f"{self.base_url}/download/{report_id}"
        headers = {"Authorization": f"Bearer {self.api_key}"}
        
        response = requests.get(url, headers=headers)
        if response.status_code == 200:
            with open(save_path, 'wb') as f:
                f.write(response.content)
            return True
        return False
```

## 七、研报分析框架

### 信息提取
```python
import re
import pdfplumber

def extract_report_info(pdf_path):
    """提取研报关键信息"""
    with pdfplumber.open(pdf_path) as pdf:
        text = ""
        for page in pdf.pages:
            text += page.extract_text() + "\n"
    
    # 提取关键信息
    info = {
        'title': extract_title(text),
        'authors': extract_authors(text),
        'date': extract_date(text),
        'institution': extract_institution(text),
        'summary': extract_summary(text),
        'keywords': extract_keywords(text)
    }
    
    return info

def extract_title(text):
    """提取标题"""
    lines = text.split('\n')
    if lines:
        return lines[0].strip()
    return None

def extract_summary(text):
    """提取摘要/核心观点"""
    # 查找"核心观点"、"摘要"等关键词
    patterns = [
        r'核心观点[：:]\s*(.*?)(?=\n\n|\Z)',
        r'摘要[：:]\s*(.*?)(?=\n\n|\Z)',
        r'投资要点[：:]\s*(.*?)(?=\n\n|\Z)'
    ]
    
    for pattern in patterns:
        match = re.search(pattern, text, re.DOTALL)
        if match:
            return match.group(1).strip()
    
    return None
```
